nginx
server {
    listen 80;
    server_name sualaptopphuquoc.vn www.sualaptopphuquoc.vn;
    return 301 https://$host$request_uri; # Tự động chuyển sang HTTPS
}

server {
    listen 443 ssl http2;
    server_name sualaptopphuquoc.vn www.sualaptopphuquoc.vn;

    ssl_certificate /etc/letsencrypt/live/sualaptopphuquoc.vn/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/sualaptopphuquoc.vn/privkey.pem;

    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    client_max_body_size 50M;
    location /static/ {
        alias /var/www/sualaptopphuquoc/staticfiles/;
        expires 30d;
        access_log off;
    }

    location /media/ {
        alias /var/www/sualaptopphuquoc/media/;
        expires 30d;
        access_log off;
    }

    location / {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
    }
}
